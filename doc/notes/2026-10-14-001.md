après montage de l'interface on a une structure 
```
header > div.header-auth 
header > div.header-auth > div.auth-pending auth-form  (wrap)
header > div.header-auth > div.auth-pending auth-form > p.auth-error (si erreur)
header > div.header-auth > div.auth-pending auth-form > p.auth-success (si process register)
header > div.header-auth > div.auth-pending auth-form > input ( name = activation_token)
header > div.header-auth > div.auth-pending auth-form > button.auth-activate-submit
header > div.header-auth > div.auth-pending auth-form > button.auth-activate-submit > i.fa fa-fw fa-check
header > div.header-auth > div.auth-pending auth-form > button.auth-activate-submit > span
```


```js
    /** Message post-inscription (email à valider) - implémentation nécessaire*/
    _buildRegisterPending(message, error = null)
    {
        const wrap = create('div', { class: 'auth-pending auth-form' })
    
        if (error)
        {
            wrap.appendChild(create('p', { class: 'auth-error', text: error }))
        }
    
        wrap.appendChild(create('p', { class: 'auth-success', text: message ?? 'Vérifiez votre email.' }))
    
        wrap.appendChild(create('input', {
            type: 'text',
            name: 'activation_token',
            inputMode: 'numeric',
            autocomplete: 'one-time-code',
            maxLength: '6',
            placeholder: 'Code d\'activation (6 chiffres)',
        }))
    
        const btnActivate = create('button', { type: 'button', class: 'auth-submit auth-activate-submit' })
        btnActivate.append(
            create('i', { class: 'fa fa-fw fa-check', 'aria-hidden': 'true' }),
            create('span', { text: 'Activer le compte' }),
        )
        wrap.appendChild(btnActivate)
    
        return wrap
    }
```
